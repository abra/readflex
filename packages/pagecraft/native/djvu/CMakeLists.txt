cmake_minimum_required(VERSION 3.16)
project(pagecraft_djvu LANGUAGES CXX)
find_package(Threads REQUIRED)
file(GLOB DJVULIBRE_SOURCES CONFIGURE_DEPENDS upstream/libdjvu/*.cpp)
add_library(pagecraft_djvu SHARED pagecraft_djvu.cpp ${DJVULIBRE_SOURCES})
target_compile_features(pagecraft_djvu PRIVATE cxx_std_17)
target_include_directories(pagecraft_djvu PRIVATE . upstream/libdjvu)
target_compile_definitions(pagecraft_djvu PRIVATE HAVE_CONFIG_H=1)
target_link_libraries(pagecraft_djvu PRIVATE Threads::Threads)
if(APPLE)
  target_link_libraries(pagecraft_djvu PRIVATE "-framework CoreFoundation")
endif()
set_target_properties(pagecraft_djvu PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
